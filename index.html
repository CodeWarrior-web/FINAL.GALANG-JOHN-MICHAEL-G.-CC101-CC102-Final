Java Code – Console-Based Program (CC102 PeTa)

Final Performance Task: Gasoline Tracker developed in CC102.

import java.util.Scanner;

public class GasolineTracker {

    public static void main(String[] args) {

        Scanner input = new Scanner(System.in);
        double[] fuel = new double[10];

        // Input 10 vehicles consumed oil
        System.out.println("Enter fuel consumption (in liters) for 10 vehicles:");
        for (int Mantika = 0; Mantika < 10; Mantika++) {
            do {
                System.out.print("Vehicle " + (Mantika + 1) + ": ");
                fuel[Mantika] = input.nextDouble();
                if (fuel[Mantika] < 0) {
                    System.out.println("Invalid! Please enter a nonnegative value.");
                }
            } while (fuel[Mantika] < 0);
        }

        // Calculate the Total oil consumed and the average of it
        double total = 0;
        for (int Mantika = 0; Mantika < 10; Mantika++) {
            total += fuel[Mantika];
        }
        double average = total / 10;

        // Find the highest and lowest
        double highest = fuel[0];
        double lowest = fuel[0];
        for (int Mantika = 1; Mantika < 10; Mantika++) {
            if (fuel[Mantika] > highest) {
                highest = fuel[Mantika];
            }
            if (fuel[Mantika] < lowest) {
                lowest = fuel[Mantika];
            }
        }

        // Count vehicles more than 20 liters of oil
        int count = 0;
        for (int Mantika = 0; Mantika < 10; Mantika++) {
            if (fuel[Mantika] > 20) {
                count++;
            }
        }

        // Display the results of oil related stuff or smth
        System.out.println("Total fuel consumption: " + total + " liters");
        System.out.println("Average fuel consumption: " + average + " liters");
        System.out.println("Highest consumption: " + highest + " liters");
        System.out.println("Lowest consumption: " + lowest + " liters");
        System.out.println("Vehicles consuming more than 20 liters: " + count);
    }
}
